cmake_minimum_required(VERSION 3.22)
project(brawl_native_core LANGUAGES CXX)
add_library(modcore STATIC src/modcore.cpp)
target_include_directories(modcore PUBLIC include)
target_compile_features(modcore PUBLIC cxx_std_17)
set_target_properties(modcore PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(MODCORE_BUILD_TESTS "Build desktop unit tests" ON)
if(MODCORE_BUILD_TESTS)
 enable_testing()
 add_executable(modcore_tests tests/core_tests.cpp)
 target_link_libraries(modcore_tests PRIVATE modcore)
 add_test(NAME core COMMAND modcore_tests)
endif()

add_executable(modcore_cli src/cli.cpp)
target_link_libraries(modcore_cli PRIVATE modcore)
